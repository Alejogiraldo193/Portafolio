Código Apps Script

Ve a Extensiones → Apps Script, pega este código y despliégalo como Web App. 

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets=['BTC','ETH','BNB','SOL','DOGE','TON','GOLD'];
  var cats={BTC:'crypto',ETH:'crypto',BNB:'crypto',
    SOL:'crypto',DOGE:'crypto',TON:'crypto',GOLD:'commodity'};
  var port = ss.getSheetByName('PORTAFOLIO');
  var data = {
    lastUpdate: Utilities.formatDate(
      new Date(),'GMT-5','dd/MM/yyyy HH:mm'),
    usdcop: port.getRange('E2').getValue(),
    assets: []
  };
  sheets.forEach(function(t) {
    var ws = ss.getSheetByName(t); if(!ws) return;
    var isBTC = (t==='BTC');
    data.assets.push({
      ticker:   t, category: cats[t],
      units:    isBTC?ws.getRange('L2').getValue()
                    :ws.getRange('K2').getValue(),
      avgBuy:   isBTC?ws.getRange('M2').getValue()
                    :ws.getRange('L2').getValue(),
      price:    isBTC?ws.getRange('N2').getValue()
                    :ws.getRange('M2').getValue(),
      invested: ws.getRange('I2').getValue()
    });
  });
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
